# Silver layer: idempotent, tested, observable
from pyspark.sql import functions as F
from delta.tables import DeltaTable

bronze = spark.readStream.table("lake.bronze.orders")

clean = (bronze
  .dropDuplicates(["order_id"])
  .withColumn("amount", F.col("amount").cast("decimal(18,2)"))
  .filter("amount >= 0"))

def upsert(df, _):
    (DeltaTable.forName(spark, "lake.silver.orders").alias("t")
      .merge(df.alias("s"), "t.order_id = s.order_id")
      .whenMatchedUpdateAll().whenNotMatchedInsertAll().execute())

clean.writeStream.foreachBatch(upsert).start()  # ✔ 99.9% pipeline reliability
